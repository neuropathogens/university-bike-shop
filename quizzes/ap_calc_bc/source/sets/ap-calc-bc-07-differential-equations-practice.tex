\section*{1\quad Modeling and verifying solutions}
\begin{setrule}R7.1–R7.2. “Proportional to $X$” means $\frac{dy}{dt}=kX$. Verify by substituting the function and its derivatives.\end{setrule}
\begin{q}{1}
A rumor spreads through a school of 500 students. The rate at which the number $y$ who have heard it increases is proportional to the product of the number who have heard it and the number who have not. Which equation models this?
\chtwo{$\dfrac{dy}{dt}=ky(500-y)$}{$\dfrac{dy}{dt}=k(500-y)$}{$\dfrac{dy}{dt}=ky$}{$\dfrac{dy}{dt}=k\dfrac{y}{500-y}$}
\end{q}
\begin{q}{2}
Which function is a solution of $y''+4y=0$?
\chfour{$e^{2x}$}{$x^2-2$}{$\sin(2x)$}{$\cos(4x)$}
\end{q}

\section*{3\quad Euler's method}
\begin{setrule}R7.4. $y_{\text{new}}=y+\Delta x\cdot\frac{dy}{dx}$ at the current point; concave up solutions are underestimated.\end{setrule}
\begin{q}{5}
Let $y=f(x)$ solve $\dfrac{dy}{dx}=2x-y$ with $f(1)=3$. Euler's method with two steps of size $0.5$ gives which approximation of $f(2)$?
\chfour{$2$}{$2.5$}{$2.75$}{$3.25$}
\end{q}
\begin{q}{6}

\chtwo{$1.1$, an underestimate}{$1.1$, an overestimate}{$1.01$, an underestimate}{$1.2$, an overestimate}
\end{q}

\section*{Free-response scoring guide}
\subsection*{Question 1: A separable differential equation (9 points)}
\textbf{(a)} At $(2,3)$: $\frac{dy}{dx}=2(3-1)=4$. Tangent: $y=3+4(x-2)$, so $f(2.1)\approx3.4$.
\begin{itemize}[leftmargin=1.6em,itemsep=0pt]
\item 1 point: tangent line $y=3+4(x-2)$
\item 1 point: approximation $f(2.1)\approx3.4$
\end{itemize}
\textbf{(b)} $\frac{d^2y}{dx^2}=(y-1)+x\frac{dy}{dx}=(y-1)+x^2(y-1)=(y-1)(1+x^2)$. At $(2,3)$ this is $2\cdot5=10>0$, and it stays positive near $x=2$ (where $y>1$), so the graph of $f$ is concave up there and lies above its tangent line: an underestimate.
\begin{itemize}[leftmargin=1.6em,itemsep=0pt]
\item 1 point: $\frac{d^2y}{dx^2}=(y-1)+x\frac{dy}{dx}$
\item 1 point: $=(y-1)(1+x^2)$, equal to 10 at $(2,3)$
\item 1 point: underestimate, because the graph is concave up near $x=2$
\end{itemize}
\textbf{(c)} $\frac{dy}{y-1}=x\,dx$, so $\ln|y-1|=\frac{x^2}{2}+C$. At $(2,3)$: $\ln2=2+C$, so $C=\ln2-2$. $y-1>0$ near the initial point, so $y-1=e^{x^2/2+\ln2-2}=2e^{x^2/2-2}$ and $y=1+2e^{x^2/2-2}$.
\begin{itemize}[leftmargin=1.6em,itemsep=0pt]
\item 1 point: separates variables
\item 1 point: antiderivatives $\ln|y-1|$ and $\frac{x^2}{2}$
\item 1 point: uses the initial condition: $\ln2=2+C$
\item 1 point: solves for $y$: $y=1+2e^{x^2/2-2}$
\end{itemize}
